# --- Directivas @ignorar: lo que el analizador descarta antes de buscar tokens ---

# descarta: espacios, tabulaciones y retornos de carro (el salto de línea lo descarta el paso 1 del algoritmo)
@ignorar [ \t\r]+

# descarta: comentarios de bloque, que empiezan con /* y terminan en el primer */ posterior; pueden ocupar varias líneas y no se anidan
@ignorar /\*[\s\S]*?\*/

# --- Funciones ---

# descripción: función que devuelve un valor; escala un número de un rango a otro
# ejemplos: map
# observaciones: se escribe exactamente como figura, en minúsculas; no puede usarse como identificador; como devuelve valor, puede usarse como operando o como sentencia.
MAP map

# descripción: función que pausa el programa durante un tiempo medido en milisegundos
# ejemplos: delay
# observaciones: se escribe exactamente como figura, en minúsculas; no puede usarse como identificador; no devuelve valor: solo puede usarse como sentencia, nunca dentro de una expresión.
DELAY delay

# descripción: función que imprime un valor por el puerto serie con salto de línea
# ejemplos: Serial.println
# observaciones: se escribe sin espacios alrededor del punto y respetando mayúsculas; no puede usarse como identificador; no devuelve valor: solo puede usarse como sentencia, nunca dentro de una expresión.
SERIAL_PRINTLN Serial\.println

# --- Signos y operadores ---

# descripción: paréntesis de apertura
# ejemplos: (
PARENTESIS_IZQ \(

# descripción: paréntesis de cierre
# ejemplos: )
PARENTESIS_DER \)

# descripción: llave de apertura, que abre el cuerpo de setup() y loop()
# ejemplos: {
LLAVE_IZQ \{

# descripción: llave de cierre, que cierra el cuerpo de setup() y loop()
# ejemplos: }
LLAVE_DER \}

# descripción: coma, separa argumentos en una llamada a función
# ejemplos: ,
COMA ,

# descripción: punto y coma, termina declaraciones, asignaciones y llamadas
# ejemplos: ;
PUNTO_Y_COMA ;

# descripción: operador de asignación
# ejemplos: =
# observaciones: solo asigna; el lenguaje no define operador de comparación, por lo que == no es un solo token.
IGUAL =

# descripción: operador de asignación con suma
# ejemplos: +=
# observaciones: se escribe pegado, sin espacio entre + y =.
SUMA_IGUAL \+=

# descripción: operador de asignación con resta
# ejemplos: -=
# observaciones: se escribe pegado, sin espacio entre - y =.
RESTA_IGUAL -=

# descripción: operador aritmético de suma
# ejemplos: +
SUMA \+

# descripción: operador aritmético de resta
# ejemplos: -
# observaciones: el lenguaje no tiene operadores unarios: -x y -5 no son expresiones; el guión solo aparece como operador binario o dentro de -=.
RESTA -

# descripción: operador aritmético de multiplicación
# ejemplos: *
# observaciones: junto con / tiene mayor precedencia que + y - (la precedencia se define en gramatica.txt).
MULTIPLICACION \*

# descripción: operador aritmético de división
# ejemplos: /
# observaciones: junto con * tiene mayor precedencia que + y - (la precedencia se define en gramatica.txt).
DIVISION /

# --- Literales ---

# descripción: cadena de texto delimitada por comillas dobles o por comillas simples
# ejemplos: "hola", 'a', ""
# observaciones: ocupa una sola línea; termina en la primera comilla igual a la de apertura; no hay secuencias de escape; puede contener el otro tipo de comilla.
CADENA "[^"\n]*"|'[^'\n]*'

# descripción: número decimal
# ejemplos: 3.25, 0.5, 195.39
# observaciones: uno o más dígitos, un punto y uno o más dígitos; sin signo ni exponente; requiere dígitos a ambos lados del punto (3. no es un decimal válido); gana sobre ENTERO porque reconoce la cadena más larga.
DECIMAL [0-9]+\.[0-9]+

# descripción: número entero
# ejemplos: 0, 42, 420
# observaciones: uno o más dígitos; sin signo ni exponente (el signo no forma parte del lexema: -5 se lee como RESTA y luego ENTERO, y no es una expresión válida).
ENTERO [0-9]+

# descripción: identificador: nombre de variable o de función definida por el programa
# ejemplos: x, cantidad, _contador, led1
# observaciones: una letra o un guion bajo seguido de cero o más letras, dígitos o guiones bajos; letras sin tilde ni eñe (solo ASCII); debe escribirse después de todas las palabras del archivo para que, en caso de empate de longitud, gane la palabra y no el identificador.
IDENTIFICADOR [a-zA-Z_][a-zA-Z0-9_]*
