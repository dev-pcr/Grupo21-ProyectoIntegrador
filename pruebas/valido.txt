/* 4. Definicion de loop(): se repite indefinidamente */
void loop() {
  /* 4.1 Expresiones aritmeticas: + - * / con precedencia y asociatividad izquierda */
  contador = 1 + 2 * 3 - 4 / 2;
  contador = (1 + 2) * (3 - 4) / 2;
  contador += temperatura * 2 + letra;
  contador -= (temperatura - 1) / 3;

  /* 4.2 Funcion con valor map(): como sentencia */
  map(contador, 0, 1023, 0, 255);

  /* 4.3 Funcion con valor map(): como operando dentro de una expresion */
  contador = map(contador, 0, 1023, 0, 255) + 1;
  contador = 2 * map(map(contador, 0, 10, 0, 100), 0, 100, 0, 1);

  /* 4.4 Funcion sin valor delay(): sentencia, con una expresion como argumento */
  delay(1000);
  delay(contador * 2 + 10);

  /* 4.5 Serial.println con una llamada a map() dentro de los argumentos */
  Serial.println("valor", map(contador, 0, 255, 0, 100));
}
